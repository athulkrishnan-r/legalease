function generateDocument() {

  const type = document.getElementById("documentType").value;
  const party = document.getElementById("partyName").value || "[Party Name]";
  const details = document.getElementById("details").value || "[Additional Details]";

  let title = "";
  let content = "";

  if (type === "rental") {

    title = "RENTAL AGREEMENT";

    content = `
RENTAL AGREEMENT

This Rental Agreement is entered into between ${party} and the other relevant party.

PROPERTY:
The property covered by this agreement shall be described as:
${details}

TERMS:
1. The tenant agrees to use the property only for lawful purposes.
2. Rent, payment dates, deposits, maintenance responsibilities and other terms shall be agreed upon by the parties.
3. Any amendments should be made in writing and agreed upon by the parties.
4. The parties agree to comply with applicable laws.

SIGNATURES:

Landlord: ______________________

Tenant: ________________________

Date: __________________________
`;

  } else if (type === "nda") {

    title = "NON-DISCLOSURE AGREEMENT";

    content = `
NON-DISCLOSURE AGREEMENT

This Non-Disclosure Agreement is made between ${party} and the receiving party.

PURPOSE:
${details}

CONFIDENTIAL INFORMATION:
Confidential information may include business information, technical information,
financial information, customer information and other information identified
as confidential by the parties.

OBLIGATIONS:
The receiving party agrees to use reasonable measures to protect confidential
information and not disclose it except as permitted by the agreement or applicable law.

TERM:
The confidentiality obligations shall continue for the period agreed by the parties.

SIGNATURES:

Disclosing Party: __________________

Receiving Party: __________________

Date: _____________________________
`;

  } else if (type === "employment") {

    title = "EMPLOYMENT AGREEMENT";

    content = `
EMPLOYMENT AGREEMENT

This Employment Agreement is between ${party} and the employee.

POSITION AND DUTIES:
${details}

The employee agrees to perform the duties reasonably assigned by the employer.

COMPENSATION:
Salary, benefits, working hours and payment arrangements shall be agreed upon
by the parties.

CONFIDENTIALITY:
The employee shall protect confidential business information.

TERMINATION:
Employment may be terminated according to the terms of this agreement and applicable law.

SIGNATURES:

Employer: _________________________

Employee: ________________________

Date: _____________________________
`;

  } else if (type === "freelance") {

    title = "FREELANCE SERVICES AGREEMENT";

    content = `
FREELANCE SERVICES AGREEMENT

This agreement is between ${party} and the freelance service provider.

SERVICES:
${details}

PAYMENT:
The client agrees to compensate the service provider according to the
payment terms agreed by the parties.

DELIVERABLES:
The service provider shall provide the agreed deliverables within the
agreed timeframe.

INTELLECTUAL PROPERTY:
Ownership and licensing of work products shall be determined by the parties
and documented in accordance with applicable law.

SIGNATURES:

Client: ___________________________

Freelancer: _______________________

Date: _____________________________
`;

  } else if (type === "privacy") {

    title = "PRIVACY POLICY";

    content = `
PRIVACY POLICY

This Privacy Policy explains how ${party} may collect, use and protect information.

INFORMATION COLLECTION:
${details}

USE OF INFORMATION:
Information may be used to provide services, improve the website,
communicate with users and comply with legal obligations.

DATA SECURITY:
Reasonable security measures should be implemented to protect personal information.

USER RIGHTS:
Users may have rights concerning access, correction, deletion or other
processing of their personal information depending on applicable law.

CONTACT:
For privacy questions, users may contact the organization through its designated contact method.
`;

  } else {

    title = "TERMS AND CONDITIONS";

    content = `
TERMS AND CONDITIONS

These Terms and Conditions govern the use of services provided by ${party}.

SERVICES:
${details}

USER RESPONSIBILITIES:
Users agree to use the service lawfully and responsibly.

INTELLECTUAL PROPERTY:
Content and intellectual property rights shall remain with their respective owners
unless otherwise stated.

LIMITATION OF LIABILITY:
Any limitations must comply with applicable law.

CHANGES:
The terms may be updated from time to time, subject to applicable requirements.

CONTACT:
Questions regarding these Terms may be directed to the service provider.
`;
  }

  document.getElementById("documentOutput").textContent =
    title + "\n\n" + content;

  document.getElementById("result").style.display = "block";
}

function downloadDocument() {

  const text = document.getElementById("documentOutput").textContent;

  const blob = new Blob([text], {
    type: "text/plain"
  });

  const url = URL.createObjectURL(blob);

  const a = document.createElement("a");

  a.href = url;
  a.download = "LegalEase-Document.txt";

  a.click();

  URL.revokeObjectURL(url);
}
